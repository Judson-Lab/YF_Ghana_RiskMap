Reproject spatial objects

# convert case data to sf object 
cases_sf <- st_as_sf(cases, coords = c("x", "y"), crs = 4326)

# convert shapefile to correct crs for meters
ghana_sf <- st_transform(ghana_map, crs = 32630)

# now match crs of case data to shapefile
cases_sf <- st_transform(cases_sf, crs = st_crs(ghana_sf))


# verify matching crs projections
st_crs(cases_sf)
st_crs(ghana_sf)

# check bounding boxes to make sure units are correct
# Bounding boxes

# create definition for the study region

  window = ghana_owin            # study area

L function for human cases

# produce L-function with envelope
# Envelope function
lf.env <- envelope(cases_ppp, Lest, nsim = 999, correction="isotropic")

# Center it by subtracting r
lf.env$obs <- lf.env$obs - lf.env$r
lf.env$lo  <- lf.env$lo  - lf.env$r
lf.env$hi  <- lf.env$hi  - lf.env$r
lf.env$theo <- 0  # theoretical CSR line at 0

KDE Plot with human case kernel density bandwidth

#  Kernel density (sigma is in meters in this CRS)
sigma_val <-   25000 # ~ peak r from Ripley's K/L, slightly less to account for variation and oversmoothing

# increased dimensions to smooth edges in final KDE plot
dens <- density.ppp(cases_ppp, sigma = sigma_val, at = "pixels", edge = TRUE, 
                    dimyx = c(500, 500))

#  Convert to data frame for ggplot
dens_df <- as.data.frame(as.im(dens))
names(dens_df) <- c("x", "y", "density")

# remove vereda borders from shapefile 
ghana_union <- st_union(ghana_sf)

kd_plot_humans2 <- ggplot() +
  # Raster / density layer
  geom_tile(data = dens_df, aes(x = x, y = y, fill = density)) +
  scale_fill_viridis(name = "Case density", option = "viridis") +
  
  # Shapefile polygon outlines
  geom_sf(data = ghana_union, fill = NA, linewidth = 0.2) +
  
  # Human cases (blue)
  geom_sf(
    data = cases_sf,
    aes(color = "Human cases"),
    shape = 16, size = 1
  ) +
  
  # Labels
  labs(
    color = "Points"
  ) +
  
  # Colorblind-safe colors
  scale_color_manual(
    values = c(
      "Human cases" = "orange"
    )
  ) +
  theme_minimal(base_size = 10) +
theme(
  axis.title = element_blank(),
  axis.text  = element_blank(),
  axis.ticks = element_blank(),
  panel.grid = element_blank()
) +
  annotation_scale(
    location   = "bl",
    width_hint = 0.25,
    text_cex   = 0.55,
    line_width = 0.6,
