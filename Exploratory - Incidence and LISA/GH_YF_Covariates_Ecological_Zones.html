# Join zone names to yf shapefile, pulling the dominant ecozone in each district
# convert crs to meters
ghana_m <- st_transform(ghana_map, 32630) 

zones_m <- st_transform(zones, 32630)

intersection <- st_intersection(ghana_m, zones_m)
## Warning: attribute variables are assumed to be spatially constant throughout
## all geometries
# area of each overlap
intersection$area <- st_area(intersection)

# area by region and ecozone
area_summary <- intersection %>%
  st_drop_geometry() %>%
  group_by(New_Region, CAPTION) %>%
  summarise(total_area = sum(area), .groups = "drop")

majority_ecozone <- area_summary %>%
  group_by(New_Region) %>%
  slice_max(total_area, n = 1, with_ties = FALSE) %>%
  ungroup()

yf_sf <- yf_data %>% 
  left_join(ghana_map, by = c("OBJECTID", "District"))

yf_final <- yf_sf %>% 
  left_join(majority_ecozone, by = "New_Region") %>% 
  mutate(zone_final = case_when(
    CAPTION == "Decidous forest" ~ "Deciduous Forest",
    CAPTION == "Transitional zone" ~ "Transitional Zone",
    CAPTION == "Coastal savanna" ~ "Coastal Savanna",
    CAPTION == "Wet evergreen" ~ "Wet Evergreen",
    CAPTION == "Guinea savanna" ~ "Guinea Savanna",
    TRUE ~ CAPTION))

zones <- yf_final %>% 
  group_by(District) %>% 
  slice_head(n = 1) %>% 
  dplyr::select(District, zone_final)

write.csv(zones, "ecological_zone_district.csv")
zone_cov <- yf_final %>% 
  group_by(zone_final, year) %>% 
  summarise(
    total_cases = sum(cases),
    cases_present = case_when(
      total_cases > 0 ~ 1,
      TRUE ~ 0
    ),
    vax_lag12 = mean(vax_lag12),
    mean_temp = mean(tmean_coldest_month),
    mean_rain = mean(precip_sum),
    mean_ndvi = mean(ndvi),
    mean_built = mean(prop_urban_and_built_up),
    .groups = "drop"
) 

zone_long <- zone_cov %>%
  pivot_longer(
    cols = -c(zone_final, year),
    names_to = "covariate",
    values_to = "value"
  )

cols <- c("Transitional Zone" = "#5d2f89", "Deciduous Forest" = "#809D5B", 
          "Coastal Savanna" = "#6CACE4","Wet Evergreen"  = "#39C7B2",
          "Guinea Savanna" = "#F19425")

zone_cov_st <- zone_long %>% 
  filter(covariate == "mean_temp" | covariate == "mean_rain" | covariate == "mean_ndvi" |
          covariate == "vax_lag12" | covariate == "mean_built") %>% 
    mutate(covariate_name = case_when(
    covariate == "mean_temp" ~ "Mean Temperature of Coldest Month",
    covariate == "mean_rain" ~ "Annual Rainfall",
    covariate == "mean_ndvi" ~ "NDVI",
    covariate == "vax_lag12" ~ "Vaccine Coverage (1-year lag)",
    covariate == "mean_built" ~ "Proportion of Built/Urban Area"
  ))


covariate_by_zone <- function(data, covariate) {
  
  df <- data[data$covariate_name == covariate, ]
  
  ggplot(df, aes(
    x = year,
    y = value,
    color = zone_final
  )) +
    geom_line(linewidth = 0.8) +
    labs(
      x = "Year",
      y = covariate,
      color = "Ecological Zone"
    ) +
    scale_color_manual(values = cols) +
    theme_minimal(base_size = 12) +
    theme(
      strip.text = element_text(face = "bold"),
      legend.position = "bottom",
       plot.background = element_rect(fill = "transparent", color = NA),
    panel.background = element_rect(fill = "transparent", color = NA)
    )
}
